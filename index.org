// Profile photo: grows out of the avatar into a large view; click anywhere or Esc shrinks it back.
(function () {
  var thumb = document.querySelector('.profile-zoom');
  var dialog = document.querySelector('.photo-dialog');
  var big = dialog.querySelector('img');
  var reduce = matchMedia('(prefers-reduced-motion: reduce)');
  function morph(reverse) {
    var a = thumb.getBoundingClientRect(), b = big.getBoundingClientRect();
    var dx = a.left + a.width / 2 - (b.left + b.width / 2);
    var dy = a.top + a.height / 2 - (b.top + b.height / 2);
    var frames = [
      { transform: 'translate(' + dx + 'px, ' + dy + 'px) scale(' + a.width / b.width + ')', borderRadius: '50%' },
      { transform: 'none', borderRadius: '24px' }
    ];
    if (reverse) frames.reverse();
    return big.animate(frames, { duration: reduce.matches ? 0 : 450, easing: 'cubic-bezier(.2, .9, .25, 1)' });
  }
  thumb.addEventListener('click', function () {
    dialog.classList.remove('is-closing');
    dialog.showModal();
    thumb.classList.add('is-open');
    morph(false);
  });
  function close() {
    if (dialog.classList.contains('is-closing')) return;
    dialog.classList.add('is-closing');
    morph(true).onfinish = function () {
      dialog.close();
      dialog.classList.remove('is-closing');
      thumb.classList.remove('is-open');
      thumb.focus({ preventScroll: true });
    };
  }
  dialog.addEventListener('click', close);
  dialog.addEventListener('cancel', function (e) { e.preventDefault(); close(); });
})();
</script>

// Emacs course: thumbnail until played, then a YouTube embed. Episode buttons swap the video.
(function () {
  var course = document.querySelector('.course');
  var player = course.querySelector('.course-player');
  var now = course.querySelector('.course-now');
  var current = course.querySelector('.course-episodes [aria-current]');
  function embed(btn) {
    var f = document.createElement('iframe');
    f.src = 'https://www.youtube-nocookie.com/embed/' + btn.dataset.id + '?autoplay=1&rel=0';
    f.title = 'Episode ' + btn.textContent + ': ' + btn.dataset.title;
    f.allow = 'autoplay; encrypted-media; picture-in-picture; fullscreen';
    f.allowFullscreen = true;
    player.replaceChildren(f);
  }
  function select(btn) {
    current.removeAttribute('aria-current');
    btn.setAttribute('aria-current', 'true');
    current = btn;
    now.children[0].textContent = btn.textContent;
    now.childNodes[1].textContent = ' ' + btn.dataset.title + ' ';
    now.children[1].textContent = btn.dataset.len;
    var play = player.querySelector('.course-play');
    if (play) {
      play.querySelector('img').src = 'https://i.ytimg.com/vi/' + btn.dataset.id + '/hqdefault.jpg';
      play.setAttribute('aria-label', 'Play episode ' + btn.textContent);
    } else {
      embed(btn);
    }
  }
  player.addEventListener('click', function (e) { if (e.target.closest('.course-play')) embed(current); });
  course.querySelector('.course-episodes').addEventListener('click', function (e) {
    var btn = e.target.closest('button');
    if (btn && btn !== current) select(btn);
  });
})();
</script>

My secondary GitHub, [[https://github.com/el-sloppo][@el-sloppo]], is where I push my favorite AI slop projects. Most of them live under [[https://github.com/emacs-os][Emacs-OS]]:

It's a systemd-inspired service supervisor written in Emacs Lisp, with a dependency graph, targets, an ~M-x elinit~ dashboard and an ~elinitctl~ CLI, plus an optional static-build patchset that turns Emacs into an actual init. It's the spiritual sequel to systemE and the core of Emacs-OS.
